# Compila el runtime + el código generado por `stk recomp` y el ejecutable soultrails_port.
cmake_minimum_required(VERSION 3.16)
project(soultrails_port C)

set(ST_RECOMP_DIR "${CMAKE_CURRENT_LIST_DIR}/../build/recomp" CACHE PATH "Salida de stk recomp")
# Nivel de optimización (stk recomp --opt): fast compila mucho antes, release es /O2 (-O2)
set(ST_OPT "release" CACHE STRING "fast | medium | release")
if(ST_OPT STREQUAL "fast")
  if(MSVC)
    set(CMAKE_C_FLAGS_RELEASE "/Od /DNDEBUG")
  else()
    set(CMAKE_C_FLAGS_RELEASE "-O0 -DNDEBUG")
  endif()
elseif(ST_OPT STREQUAL "medium")
  if(MSVC)
    set(CMAKE_C_FLAGS_RELEASE "/O1 /DNDEBUG")
  else()
    set(CMAKE_C_FLAGS_RELEASE "-O1 -DNDEBUG")
  endif()
endif()
message(STATUS "Optimización del código recompilado: ${ST_OPT} (${CMAKE_C_FLAGS_RELEASE})")
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../runtime ${CMAKE_BINARY_DIR}/runtime)
if(NOT EXISTS "${ST_RECOMP_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "No encuentro ${ST_RECOMP_DIR}. Ejecuta antes: python stk.py recomp")
endif()
add_subdirectory(${ST_RECOMP_DIR} ${CMAKE_BINARY_DIR}/recomp)
option(ST_MELONDS "Port híbrido: ARM7, sonido, guardado, cartucho y gráficos con melonDS (GPL-3.0)" OFF)
if(ST_MELONDS)
  # melonDS no compila con MSVC (usa __int128 y __builtin_*): en Windows, clang-cl (-T ClangCL)
  if(MSVC AND NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "melonDS necesita clang-cl en Windows. Instala el componente 'Compilador C++ Clang para "
                        "Windows' de Visual Studio Build Tools y vuelve a ejecutar stk (usa -T ClangCL).")
  endif()
  enable_language(CXX)
  include(FetchContent)
  if(POLICY CMP0169)
    cmake_policy(SET CMP0169 OLD)   # FetchContent_Populate directo: se parchea antes de configurar
  endif()
  set(ST_MELONDS_SOURCE "" CACHE PATH "Copia local de melonDS 1.0 (si se deja vacío, se descarga)")
  if(ST_MELONDS_SOURCE)
    set(FETCHCONTENT_SOURCE_DIR_MELONDS "${ST_MELONDS_SOURCE}")
  endif()
  FetchContent_Declare(melonds GIT_REPOSITORY https://github.com/melonDS-emu/melonDS.git GIT_TAG 1.0 GIT_SHALLOW TRUE)
  FetchContent_GetProperties(melonds)
  if(NOT melonds_POPULATED)
    FetchContent_Populate(melonds)
  endif()
  execute_process(COMMAND ${CMAKE_COMMAND} -DMELON_SRC=${melonds_SOURCE_DIR}/src
                          -P ${CMAKE_CURRENT_LIST_DIR}/melon/patch_melonds.cmake RESULT_VARIABLE st_patch_rc)
  if(st_patch_rc)
    message(FATAL_ERROR "No se pudo parchear melonDS")
  endif()
  set(BUILD_QT_SDL OFF CACHE BOOL "" FORCE)
  set(ENABLE_JIT OFF CACHE BOOL "" FORCE)
  set(ENABLE_GDBSTUB OFF CACHE BOOL "" FORCE)
  set(ENABLE_OGLRENDERER OFF CACHE BOOL "" FORCE)
  set(ENABLE_LTO_RELEASE OFF CACHE BOOL "" FORCE)
  add_subdirectory(${melonds_SOURCE_DIR} ${melonds_BINARY_DIR} EXCLUDE_FROM_ALL)
  if(MSVC)   # clang-cl: melonDS solo se compila oficialmente con GCC/Clang de MinGW en Windows
    target_compile_definitions(core PRIVATE strncasecmp=_strnicmp strcasecmp=_stricmp _CRT_SECURE_NO_WARNINGS)
    target_include_directories(core PRIVATE ${CMAKE_CURRENT_LIST_DIR}/melon/compat)
    target_compile_options(core PRIVATE -Wno-deprecated-declarations -Wno-unused-command-line-argument)
    target_compile_options(teakra PRIVATE -Wno-unused-command-line-argument)
  endif()
  add_executable(soultrails_port ${CMAKE_CURRENT_LIST_DIR}/main.c ${CMAKE_CURRENT_LIST_DIR}/melon/st_melon.cpp
                 ${CMAKE_CURRENT_LIST_DIR}/melon/melon_platform.cpp)
  set_target_properties(soultrails_port PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
  target_compile_definitions(soultrails_port PRIVATE ST_MELONDS)
  target_include_directories(soultrails_port PRIVATE ${CMAKE_CURRENT_LIST_DIR})
  target_link_libraries(soultrails_port PRIVATE st_recomp_generated core)
  if(MSVC)
    target_compile_definitions(soultrails_port PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
else()
  add_executable(soultrails_port ${CMAKE_CURRENT_LIST_DIR}/main.c)
  target_link_libraries(soultrails_port PRIVATE st_recomp_generated)
endif()
if(NOT WIN32)
  find_package(Threads REQUIRED)
  target_link_libraries(soultrails_port PRIVATE Threads::Threads)
endif()
